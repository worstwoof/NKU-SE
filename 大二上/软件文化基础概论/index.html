function AI_Decide() {
  candidates = getNeighbors(distance=2); // 剪枝
  for point in candidates:
    myScore = evaluate(point, AI);
    oppScore = evaluate(point, Human);
    total = myScore + (oppScore * 0.8);
    if total > bestScore:
      bestScore = total; bestPoint = point;
  placePiece(bestPoint);
}
// 核心评估函数：检查四个方向
function evaluatePoint(board, idx, color) {
  const dirs = [[1,0], [0,1], [1,1], [1,-1]];
  let totalScore = 0;

  for (let d of dirs) {
    let count = 1, blocked = 0;
    // 正向搜索
    for (let k=1; k<=4; k++) {
      let p = getPiece(idx + k*d);
      if (p == color) count++;
      else if (p == null) break;
      else { blocked++; break; }
    }
    // 反向搜索 (代码省略...)
    
    // 评分表 (Heuristic Table)
    if (count >= 5) totalScore += 100000;
    else if (count == 4 && blocked == 0) totalScore += 10000;
    else if (count == 4 && blocked == 1) totalScore += 1000;
    else if (count == 3 && blocked == 0) totalScore += 1000;
    else if (count == 2 && blocked == 0) totalScore += 100;
  }
  return totalScore;
}

2. 核心算法：极大极小值

AI 假设对手（玩家）也是极其聪明的：

  • MAX 层 (AI): 尝试找到让自己得分最高的一步。
  • MIN 层 (Human): AI 预测对手会走让自己得分最低（即对手得分最高）的一步。

4. 算法优化：邻域剪枝

为了在浏览器端实现 100ms 内的响应，我们不扫描全盘 225 个点。

  • 逻辑： 好棋通常下在已有棋子的周围。
  • 实现： 仅搜索距离现有棋子 Distance <= 2 的空位。
  • 效果： 将每层搜索节点从 225 降低至 20~30 个，计算效率提升 10倍 以上。
